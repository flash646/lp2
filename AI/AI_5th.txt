# Simple Chatbot for Customer Interaction
# College Practice Program

print("===== Welcome to ABC Store Chatbot =====")
print("Type 'bye' to exit the chat.\n")

while True:

    user = input("You: ").lower()

    # Exit condition
    if user == "bye":
        print("Chatbot: Thank you for visiting ABC Store!")
        break

    # Greetings
    elif user in ["hi", "hello", "hey"]:
        print("Chatbot: Hello! How can I help you today?")

    # Product inquiry
    elif "product" in user:
        print("Chatbot: We sell laptops, mobiles, and headphones.")

    # Price inquiry
    elif "price" in user:
        print("Chatbot: Prices start from Rs. 500 onwards.")

    # Delivery inquiry
    elif "delivery" in user:
        print("Chatbot: Delivery usually takes 3 to 5 days.")

    # Contact inquiry
    elif "contact" in user:
        print("Chatbot: You can contact us at 9876543210.")

    # Thank you response
    elif "thank" in user:
        print("Chatbot: You're welcome!")

    # Default response
    else:
        print("Chatbot: Sorry, I did not understand that.")
