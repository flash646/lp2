/*Write a Java/C/C++/Python program that contains a string (char pointer) with a value \Hello 
World9. The program should AND or and XOR each character in this string with 127 and display 
the result. */

class BitwiseString {
    public static void main(String[] args) {
        String str = "\\Hello World9";

        System.out.println("Original String: " + str);

        System.out.println("\nAfter AND with 127:");
        for (int i = 0; i < str.length(); i++) {
            char result = (char)(str.charAt(i) & 127);
            System.out.print(result);
        }

        System.out.println("\n\nAfter XOR with 127:");
        for (int i = 0; i < str.length(); i++) {
            int value = (char)(str.charAt(i) ^ 127);
            if (value >= 32 && value <= 126) {
                System.out.println(value + " -> " + (char)value);
            } else {
                System.out.println(value + " -> non-printable");
            }
            //System.out.print((char)value);
        }
        System.out.println();
    }
}
