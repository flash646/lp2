/*Write a Java/C/C++/Python program to perform encryption and decryption using the method of 
the method of Transposition technique. */

import java.util.Scanner;

public class Main {

    static String text = "";
    static String cipher = "";
    static int key = 0;

    // Encryption
    static String encrypt(String text, int key) {
        StringBuilder cipher = new StringBuilder();

        for (int col = 0; col < key; col++) {
            for (int i = col; i < text.length(); i += key) {
                cipher.append(text.charAt(i));
            }
        }
        return cipher.toString();
    }

    // Decryption
    static String decrypt(String cipher, int key) {
        char[] text = new char[cipher.length()];
        int index = 0;

        for (int col = 0; col < key; col++) {
            for (int i = col; i < cipher.length(); i += key) {
                text[i] = cipher.charAt(index++);
            }
        }
        return new String(text);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        while (true) {
            System.out.println("\nChoose an option:");
            System.out.println("1) Enter Plain Text");
            System.out.println("2) Enter Key");
            System.out.println("3) Encrypt");
            System.out.println("4) Decrypt");
            System.out.println("5) Exit");

            int choice = sc.nextInt();
            sc.nextLine(); // consume newline

            switch (choice) {
                case 1:
                    System.out.print("Enter plain text: ");
                    text = sc.nextLine();
                    break;

                case 2:
                    System.out.print("Enter key (numeric): ");
                    key = sc.nextInt();
                    break;

                case 3:
                    if (text.isEmpty() || key <= 0) {
                        System.out.println("Please enter valid text and key first.");
                    } else {
                        cipher = encrypt(text, key);
                        System.out.println("Encrypted Text: " + cipher);
                    }
                    break;

                case 4:
                    if (cipher.isEmpty() || key <= 0) {
                        System.out.println("No cipher available. Encrypt first.");
                    } else {
                        String decrypted = decrypt(cipher, key);
                        System.out.println("Decrypted Text: " + decrypted);
                    }
                    break;

                case 5:
                    System.out.println("Exiting...");
                    sc.close();
                    return;

                default:
                    System.out.println("Invalid choice.");
            }
        }
    }
}
