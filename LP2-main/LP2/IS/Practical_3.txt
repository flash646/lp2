/*Write a Java/C/C++/Python program to implement DES algorithm.*/

import javax.crypto.Cipher;
import javax.crypto.SecretKey;
import javax.crypto.SecretKeyFactory;
import javax.crypto.spec.DESKeySpec;
import java.util.Base64;

public class Main {
    public static void main(String[] args) throws Exception {
        String plainText = "HELLODES";
        String keyStr = "12345678"; // DES key must be 8 bytes

        DESKeySpec keySpec = new DESKeySpec(keyStr.getBytes());
        SecretKeyFactory keyFactory = SecretKeyFactory.getInstance("DES");
        SecretKey key = keyFactory.generateSecret(keySpec);

        Cipher cipher = Cipher.getInstance("DES");

        // Encryption
        cipher.init(Cipher.ENCRYPT_MODE, key);
        byte[] encrypted = cipher.doFinal(plainText.getBytes());
        String encText = Base64.getEncoder().encodeToString(encrypted);
        System.out.println("Encrypted: " + encText);

        // Decryption
        cipher.init(Cipher.DECRYPT_MODE, key);
        byte[] decrypted = cipher.doFinal(Base64.getDecoder().decode(encText));
        System.out.println("Decrypted: " + new String(decrypted));
    }
}
