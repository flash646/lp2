import heapq

def a_star_search(graph, heuristics, start, goal):

    # Priority Queue -> (f(n), g(n), current_node, path)
    open_list = [(heuristics[start], 0, start, [start])]

    visited = set()

    while open_list:

        f, g, current, path = heapq.heappop(open_list)

        # Goal check
        if current == goal:
            return path, g

        visited.add(current)

        # Visit neighbours
        for neighbor, cost in graph[current]:

            if neighbor not in visited:

                # Calculate g(n)
                g_new = g + cost

                # Calculate f(n)
                f_new = g_new + heuristics[neighbor]

                # Push into priority queue
                heapq.heappush(
                    open_list,
                    (f_new, g_new, neighbor, path + [neighbor])
                )

    return None, float('inf')


def main():

    graph = {}
    heuristics = {}

    # Number of nodes
    n = int(input("Enter number of nodes : "))

    # Graph input
    for i in range(n):

        node = input("\nEnter node name : ")

        graph[node] = []

        h = int(input("Enter heuristic value for {} : ".format(node)))

        heuristics[node] = h

        edges = int(input("Enter number of neighbours for {} : ".format(node)))

        for j in range(edges):

            neighbor = input("Enter neighbour {} : ".format(j + 1))

            cost = int(input("Enter cost to reach {} : ".format(neighbor)))

            graph[node].append((neighbor, cost))

    # Start and Goal
    start = input("\nEnter start node : ")
    goal = input("Enter goal node : ")

    # Call A*
    path, cost = a_star_search(graph, heuristics, start, goal)

    # Output
    if path:
        print("\nShortest Path Found : ", " -> ".join(path))
        print("Total Cost : ", cost)
    else:
        print("Path not found")


if __name__ == "__main__":
    main()
